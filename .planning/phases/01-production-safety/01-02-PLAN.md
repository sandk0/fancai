```python
from sentry_sdk.integrations.fastapi import FastApiIntegration
from sentry_sdk.integrations.starlette import StarletteIntegration
from sentry_sdk.integrations.celery import CeleryIntegration
from .config import settings

def init_sentry() -> None:
    """Initialize Sentry SDK if DSN is configured.

    Silently returns if SENTRY_DSN is not set — Sentry is optional.
    Per user decision: 100% sample rate (self-hosted, no limits),
    errors + performance traces enabled.
    """
    dsn = getattr(settings, "SENTRY_DSN", None) or ""
    if not dsn:
        return

    sentry_sdk.init(
        dsn=dsn,
        environment="production" if not settings.DEBUG else "development",
        release=f"fancai@{settings.APP_VERSION}",
        traces_sample_rate=1.0,  # 100% — self-hosted, no quotas
        send_default_pii=False,
        integrations=[
            StarletteIntegration(transaction_style="endpoint"),
            FastApiIntegration(transaction_style="endpoint"),
            CeleryIntegration(monitor_beat_tasks=True),
        ],
        attach_stacktrace=True,
    )
```

**DEPLOY-04 — Database backup strategy:**

Per user decisions: Daily pg_dump at 03:00 MSK via cron on host, compressed with gzip, uploaded to Yandex Object Storage (S3-compatible), 7-day retention, backup failures reported to Sentry.

1. Create `scripts/backup-postgres.sh`:
```bash
#!/bin/bash
# PostgreSQL backup script for fancai
# Runs via cron on host: 0 3 * * * /path/to/backup-postgres.sh
# Requires: pg_dump (from postgresql-client), aws-cli, curl
set -euo pipefail

TIMESTAMP=$(date +%Y%m%d-%H%M%S)
BACKUP_FILE="/tmp/fancai-backup-${TIMESTAMP}.sql.gz"

# Required environment variables
S3_BUCKET="${S3_BACKUP_BUCKET:?S3_BACKUP_BUCKET is required}"
S3_ENDPOINT="${AWS_ENDPOINT_URL:-https://storage.yandexcloud.net}"
DB_HOST="${DB_HOST:-postgres}"
DB_PORT="${DB_PORT:-5432}"
DB_USER="${DB_USER:-postgres}"
DB_NAME="${DB_NAME:-bookreader_dev}"

# Optional: Sentry DSN for failure reporting
SENTRY_DSN="${SENTRY_DSN:-}"

report_error() {
    local msg="$1"
    echo "[ERROR] $(date): $msg" >&2
    if [ -n "$SENTRY_DSN" ]; then
        # Report to Sentry via curl (no SDK needed in bash)
        curl -s -X POST "$SENTRY_DSN" \
            -H "Content-Type: application/json" \
            -d "{\"message\": \"Backup failed: $msg\", \"level\": \"error\"}" \
            2>/dev/null || true
    fi
}

cleanup() {
    rm -f "${BACKUP_FILE}" 2>/dev/null || true
}
trap cleanup EXIT

# Step 1: Dump and compress
echo "[$(date)] Starting backup: ${DB_NAME}@${DB_HOST}:${DB_PORT}"
PGPASSWORD="${DB_PASSWORD:?DB_PASSWORD is required}" pg_dump \
    -h "${DB_HOST}" \
    -p "${DB_PORT}" \
    -U "${DB_USER}" \
    -d "${DB_NAME}" \
    --no-owner --no-acl \
    | gzip > "${BACKUP_FILE}" || {
    report_error "pg_dump failed for ${DB_NAME}"
    exit 1
}

BACKUP_SIZE=$(du -h "${BACKUP_FILE}" | cut -f1)
echo "[$(date)] Dump complete: ${BACKUP_SIZE}"

# Step 2: Upload to S3
aws s3 cp "${BACKUP_FILE}" "s3://${S3_BUCKET}/backups/${TIMESTAMP}.sql.gz" \
    --endpoint-url "${S3_ENDPOINT}" || {
    report_error "S3 upload failed for ${TIMESTAMP}.sql.gz"
    exit 1
}
echo "[$(date)] Uploaded to s3://${S3_BUCKET}/backups/${TIMESTAMP}.sql.gz"

# Step 3: Delete backups older than 7 days
CUTOFF_DATE=$(date -d "7 days ago" +%Y%m%d 2>/dev/null || date -v-7d +%Y%m%d 2>/dev/null || echo "")
if [ -n "$CUTOFF_DATE" ]; then
    aws s3 ls "s3://${S3_BUCKET}/backups/" --endpoint-url "${S3_ENDPOINT}" 2>/dev/null \
        | awk '{print $4}' \
        | while read -r file; do
            file_date=$(echo "$file" | grep -oP '^\d{8}' || echo "")
            if [ -n "$file_date" ] && [ "$file_date" -lt "$CUTOFF_DATE" ] 2>/dev/null; then
                aws s3 rm "s3://${S3_BUCKET}/backups/$file" --endpoint-url "${S3_ENDPOINT}"
                echo "[$(date)] Deleted old backup: $file"
            fi
        done
fi

echo "[$(date)] Backup complete: ${TIMESTAMP}.sql.gz (${BACKUP_SIZE})"
```
