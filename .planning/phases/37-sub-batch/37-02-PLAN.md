    ```python
    def process_batch_results(
        batch_results: list[dict],
        batch_chapters: list,
    ) -> tuple[list[tuple], list]:
        """Обработка результатов batch extraction (D-18).

        Разделяет результаты на successful и failed.

        Args:
            batch_results: [{result, metrics, chapter_idx, truncated_text?}] из extract_chapters_batch
            batch_chapters: исходные Chapter объекты в том же порядке

        Returns:
            (successful: [(chapter, result_dict)], failed: [chapter])
            result_dict содержит: result (ChapterAnalysisResult), metrics, raw_result (original dict)
        """
        successful = []
        failed = []

        for item in batch_results:
            ch = batch_chapters[item["chapter_idx"]]
            if item["result"] is None:
                # Truncated or failed -> retry individual
                failed.append(ch)
            else:
                chapter_result = modal_response_to_chapter_result({"result": item["result"], "metrics": item["metrics"]})
                successful.append((ch, {
                    "analysis": chapter_result,
                    "metrics": item["metrics"],
                    "raw_result": item["result"],
                }))

        return successful, failed
    ```
