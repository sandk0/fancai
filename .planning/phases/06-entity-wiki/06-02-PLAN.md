    **Инвариант 1: _get_current_milestone — milestones не утекают из будущих глав**
    ```python
    @given(
        milestones=st.lists(
            st.fixed_dictionaries({
                "up_to_chapter": st.integers(min_value=0, max_value=200),
                "biography": st.text(min_size=1, max_size=50),
            }),
            min_size=0, max_size=20
        ),
        current_chapter=st.integers(min_value=0, max_value=200),
    )
    @settings(max_examples=500)
    def test_milestone_never_leaks_future(milestones, current_chapter):
        result = EntityService._get_current_milestone(milestones, current_chapter)
        if result is not None:
            assert result["up_to_chapter"] <= current_chapter
    ```

    **Инвариант 2: _filter_events_by_chapter — events не утекают**
    ```python
    @given(
        events=st.lists(
            st.fixed_dictionaries({
                "chapter_number": st.integers(min_value=0, max_value=200),
                "event_action": st.text(min_size=1, max_size=50),
                "event_inner_state": st.one_of(st.none(), st.text(max_size=50)),
            }),
            min_size=0, max_size=30
        ),
        current_chapter=st.integers(min_value=0, max_value=200),
    )
    @settings(max_examples=500)
    def test_events_never_leak_future(events, current_chapter):
        result = EntityService._filter_events_by_chapter(events, current_chapter)
        for event in result:
            assert event["chapter_number"] <= current_chapter
    ```

    **Инвариант 3: _filter_aliases_from_raw — aliases не утекают**
    ```python
    @given(
        aliases=st.lists(
            st.fixed_dictionaries({
                "name": st.text(min_size=1, max_size=30),
                "reveal_chapter": st.one_of(st.none(), st.integers(min_value=0, max_value=200)),
            }),
            min_size=0, max_size=20
        ),
        current_chapter=st.integers(min_value=0, max_value=200),
    )
    @settings(max_examples=500)
    def test_aliases_never_leak_future(aliases, current_chapter):
        result = EntityService._filter_aliases_from_raw(aliases, current_chapter)
        # Каждый видимый alias должен иметь reveal_chapter <= current_chapter или None
        for alias_data in aliases:
            name = alias_data.get("name", "")
            reveal = alias_data.get("reveal_chapter")
            if name in result:
                assert reveal is None or reveal <= current_chapter
    ```

    **Инвариант 4: _get_current_relationship_milestone — relationship milestones не утекают**
    Аналогичен Инварианту 1 но для _get_current_relationship_milestone.

    **Инвариант 5: Монотонность — больше прочитанных глав = больше или равно видимых данных**
    ```python
    @given(
        events=st.lists(...),
        ch_a=st.integers(min_value=0, max_value=100),
        ch_b=st.integers(min_value=0, max_value=100),
    )
    def test_events_monotonic(events, ch_a, ch_b):
        assume(ch_a <= ch_b)
        result_a = EntityService._filter_events_by_chapter(events, ch_a)
        result_b = EntityService._filter_events_by_chapter(events, ch_b)
        assert len(result_a) <= len(result_b)
    ```

    Все тесты используют `@settings(max_examples=500)` для баланса покрытия и скорости.
    Добавить `@settings(database=None)` если нужна воспроизводимость без .hypothesis/ директории.
