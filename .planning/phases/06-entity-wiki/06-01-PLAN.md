       MAX_DEPTH = 2

       for depth in range(MAX_DEPTH):
           if len(entities) <= BATCH_SIZE:
               # Одиночный pass
               result = await self._single_reduce_pass(entities)
               await self._execute_reduce_operations(result, book_id)
               break

           # Батчевый reduce
           batches = [entities[i:i+BATCH_SIZE] for i in range(0, len(entities), BATCH_SIZE)]
           all_ops = {"merge_operations": [], "delete_operations": []}
           for batch in batches:
               result = await self._single_reduce_pass(batch)
               all_ops["merge_operations"].extend(result.get("merge_operations", []))
               all_ops["delete_operations"].extend(result.get("delete_operations", []))

           await self._execute_reduce_operations(all_ops, book_id)

           # Перезагрузить сущности для следующего раунда
